56template <typename E1, typename... Es>
57void link_error_info_helper(const error_info_base **pp, E1 &e1,
58 Es &...es) noexcept
59{
60 *pp = std::addressof(e1);
61 pp = &(*pp)->next_;
62 link_error_info_helper(pp, es...);
63}

65template <typename... ErrorInfos>
66const error_info_base *
67link_error_infos(std::tuple<ErrorInfos...> &error_infos) noexcept
68{
69 const error_info_base *r = nullptr;
70 const error_info_base **pp = &r;
71
72 std::apply([&](auto &...errors) { link_error_info_helper(pp, errors...); },
73 error_infos);

75 return r;
