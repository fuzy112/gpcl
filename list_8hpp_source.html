36template <typename T>
37struct list_node : noncopyable
38{
39 using value_type = T;
40
41 propagate_const<list_node *> prev_ = this;
42
43 propagate_const<list_node *> next_ = this;
44
45#ifdef GPCL_DEBUG
46 bool empty_ = true;
47#endif
48
49 union
50 {
51 char dummy_;
52 T val_;
53 };
54
56 constexpr list_node() noexcept : dummy_() {}
57
62 template <typename... Args>
63 explicit list_node(list_node *prev, list_node *next, Args &&...args)
64 : prev_(prev),
65 next_(next),
66#ifdef GPCL_DEBUG
67 empty_(false),
68#endif
69 val_(std::forward<Args>(args)...)
70 {
71 }
72
76 ~list_node() noexcept
77 {
78#ifdef GPCL_DEBUG
79 GPCL_ASSERT(empty_);
80#endif
81 }
82
84 void destroy() noexcept
85 {
86#ifdef GPCL_DEBUG
87 GPCL_ASSERT(!empty_);
88 empty_ = true;
89#endif
90
91 val_.~T();
92 dummy_ = '\0';
93 }
94
96 void swap(list_node &other) noexcept
97 {
98 if (this == &other)
99 return;
100
101 list_node *prev = prev_;
102 list_node *next = next_;
103
104 list_node *other_prev = other.prev_;
105 list_node *other_next = other.next_;
106
107 this->prev_ = other_prev;
108 this->next_ = other_next;
109 prev->next_ = &other;
110 next->prev_ = &other;
111
112 other_prev->next_ = this;
113 other_next->prev_ = this;
114 other.prev_ = prev;
115 other.next_ = next;
116 }
117};

354 list &operator=(const list &other)

358 {

360 {

362 auto last = end();
363 splice(first, list(other, other.get_allocator()));
364 erase(first, last);
365 p_.first() = other.p_.first();

368 p_.first() = other.p_.first();

373 auto first = begin();
374 auto last = end();

378 *first = *other_first;

703 iterator erase(const_iterator pos) noexcept
704 {
705 auto node = const_cast<node_type *>(pos.get_node());
706 node_type *prev = node->prev_;
707 node_type *next = node->next_;
708 destroy_node(node);
709 --size_;
710 prev->next_ = next;
711 next->prev_ = prev;
712 return iterator(next);
713 }

716 iterator erase(const_iterator first, const_iterator last) noexcept
717 {
718 auto node = const_cast<node_type *>(first.get_node());
719 auto next = const_cast<node_type *>(last.get_node());
720 node_type *prev = node->prev_;
721
722 while (node != next)
723 {
724 node_type *tmp = node->next_;
725 destroy_node(node);
726 node = tmp;
727 --size_;
728 }
729 prev->next_ = next;
730 next->prev_ = prev;
731 return iterator(next);
732 }

771 void splice(const_iterator pos, list &&other, const_iterator first,
772 const_iterator last) noexcept
773 {
774 GPCL_ASSERT(get_allocator() == other.get_allocator());
775
776 if (this != &other)
777 {
778 size_type count = std::distance(first, last);
779 other.size_ -= count;
780 size_ += count;
781 }
782
783 auto node = const_cast<node_type *>(first.get_node());
784 auto next = const_cast<node_type *>(last.get_node());
785 node_type *prev = node->prev_;
786 node_type *node_last = next->prev_;
787
788 prev->next_ = next;
789 next->prev_ = prev;
790
791 auto pos_node = const_cast<node_type *>(pos.get_node());
792 node_type *pos_prev_node = pos_node->prev_;
793 node->prev_ = pos_prev_node;
794 node_last->next_ = pos_node;
795
796 pos_prev_node->next_ = node;
797 pos_node->prev_ = node_last;
798 }

803 void sort()
804 {
805 auto i = cbegin();
806 auto last = cend();
807
808 while (i != last)
809 {
810 auto j = std::next(i);
811 while (j != last)
812 {
813 if (*j < *i)
814 {
815 const auto tmp = j;
816 splice(i, std::move(*this), j++);
817 i = tmp;
818 continue;
819 }
820 ++j;
821 }
822
823 ++i;
824 }
825 }
