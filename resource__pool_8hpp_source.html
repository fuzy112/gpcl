46 ~resource_pool() noexcept
47 {
48 block_info block = block_list_;
49 while (block.ptr)
50 {
51 auto next = *reinterpret_cast<block_info *>(
52 block.ptr + block_info_offset(block.size));
53 upstream_->deallocate(block.ptr, block.size);
54 block = next;
55 }
56 }

62 [[nodiscard]] void *malloc()
63 {
64 if (storage_.empty())
65 {
66 if (request_new_block(chunk_size_))
67 return storage_.malloc(chunk_size_);
68 else

90 [[nodiscard]] void *ordered_malloc(size_type n)

95 return ret;

136 struct block_info
137 {
138 // Pointer to the start of the next block.
139 char *ptr;
140
141 // Size of the next block.
142 size_type size;
143 }
144#ifdef __GNUC__
145 __attribute__((may_alias))
146#endif
147 ;
148
149 // Requests a block with at least min_size bytes for chunks.
150 bool request_new_block(size_type min_size, bool ordered = false)
151 {
152 auto sz = next_size_ * chunk_size_;
153 while (sz < min_size + sizeof(block_info) * 2)
154 sz *= 2;
155
156 next_size_ *= 2;
157
158 char *block = reinterpret_cast<char *>(upstream_->allocate(sz));
159 if (block == nullptr)
160 return false;
161
162 block_list_ = add_block(block, sz, block_list_, ordered);
163 GPCL_VERIFY_FALSE(storage_.empty());
164 return true;
165 }
166
167 block_info add_block(char *ptr, size_type sz, block_info end, bool ordered)
168 {
169 auto chunks_size = block_info_offset(sz);
170 if (ordered)
171 storage_.add_ordered_block(ptr, chunks_size, chunk_size_);
172 else
173 storage_.add_block(ptr, chunks_size, chunk_size_);
174 *reinterpret_cast<block_info *>(ptr + chunks_size) = end;
175 return {ptr, sz};
176 }
177
178 difference_type block_info_offset(size_type block_size)
179 {
180 return (block_size - sizeof(block_info)) / alignof(block_info) *
181 alignof(block_info);
182 }
183
184 size_type chunk_count(size_type n)
185 {
186 const size_type array_size = n * requested_size_;
187 const size_type ret = (array_size + chunk_size_ - 1) / chunk_size_;
188 GPCL_VERIFY(ret * chunk_size_ >= n * requested_size_);
189 return ret;
190 }
191
192 memory_resource *upstream_;
193 const size_type requested_size_;
194 const size_type chunk_size_;
195 size_type next_size_;
196 block_info block_list_;
197 simple_segregated_storage<size_type> storage_;
198};
