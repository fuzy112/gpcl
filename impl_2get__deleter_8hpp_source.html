19template <typename Deleter, typename T>
20Deleter *get_deleter(const shared_ptr<T> &p) noexcept
21{
22 if (p.use_count() == 0)
23 return nullptr;
24 auto* ty = static_cast<const type_info *>(
25 p.s_->operate(detail::ref_count_operation::get_deleter_type_info));
26 if (!ty)
27 return nullptr;
28 if (*ty != typeid_<Deleter>())
29 return nullptr;
30 return static_cast<Deleter *>(p.s_->operate(detail::ref_count_operation::get_deleter));
31}
