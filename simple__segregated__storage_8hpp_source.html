24template <typename SizeType = std::size_t, typename VoidPointer = void *>
25class simple_segregated_storage : noncopyable

41 void_pointer segregate(void_pointer const block, size_type sz,
42 size_type partition_sz,
43 void_pointer end = nullptr) noexcept
44 {
45 GPCL_ASSERT(block);
46 GPCL_ASSERT(sz >= partition_sz);
47
48 void_pointer chunk = block;
49 void_pointer block_end =
50 reinterpret_cast<char *>(block) + sz - partition_sz;
51 while (reinterpret_cast<void_pointer>(reinterpret_cast<char *>(chunk) +
52 partition_sz) < block_end)
53 {
54 next_chunk(chunk) = reinterpret_cast<char *>(chunk) + partition_sz;
55 chunk = next_chunk(chunk);
56 }
57 next_chunk(chunk) = end;
58 return block;
59 }

74 void add_block(void_pointer const block, size_type sz,
75 size_type partition_sz) noexcept
76 {
77 free_list_ = segregate(block, sz, partition_sz, free_list_);
78 GPCL_VERIFY(free_list_);
79 }

96 void add_ordered_block(void_pointer const block, size_type sz,
97 size_type partition_sz) noexcept
98 {
99 void_pointer &pos = upper_bound(block);
100 pos = segregate(block, sz, partition_sz, pos);
101 GPCL_VERIFY(free_list_);
102 }

107 bool empty() const noexcept { return free_list_ == nullptr; }

116 void_pointer malloc(size_type partition_sz) noexcept
117 {
118 (void)partition_sz;
119 GPCL_ASSERT(!empty());
120 auto ret = free_list_;
121 free_list_ = next_chunk(free_list_);
122 return ret;

136 void_pointer malloc_n(size_type n, size_type partition_sz) noexcept
137 {
138 void_pointer *p = &free_list_;
139 void_pointer *start = p;
140 size_type m = 0;
141 while (*p)
142 {
143 std::ptrdiff_t diff = reinterpret_cast<char *>(next_chunk(*p)) -
144 reinterpret_cast<char *>(*p);
145 if (diff != std::ptrdiff_t(partition_sz))
146 {
147 m = 0;
148 p = &next_chunk(*p);
149 start = p;
150 }
151 else
152 {
153 ++m;
154 p = &next_chunk(*p);
155
156 if (m == n)
157 {
158 auto ret = *start;
159 *start = *p;
160 return ret;
161 }
162 }
163 }
164
165 return nullptr;
166 }

223 bool is_free(void const *p) const
224 {
225 void_pointer iter = free_list_;
226 while (iter)
227 {
228 if (iter == p)
229 return true;
230
231 iter = next_chunk(iter);
232 }
233
234 return false;
235 }

248private:
249 void_pointer &upper_bound(void_pointer const p) noexcept
250 {
251 void_pointer *pp = &free_list_;
252 while (*pp)
253 {
254 void_pointer &next = next_chunk(*pp);
255 if (next > p)
256 {
257 return next;
258 }
259 pp = &next;
260 }
261 return *pp;
262 }
263
264 static void_pointer &next_chunk(void_pointer chunk) noexcept
265 {
266 GPCL_ASSERT(chunk);
267 return *reinterpret_cast<void_pointer *>(chunk);
268 }

270 void_pointer free_list_{};
271};
