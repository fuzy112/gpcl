48class win_stacktrace_entry
49{
50 PVOID addr_ = 0;
51
52public:
53 using native_handle_type = PVOID;
54
55 constexpr win_stacktrace_entry() noexcept = default;
56
57 constexpr win_stacktrace_entry(native_handle_type address) : addr_(address) {}
58
59 constexpr win_stacktrace_entry(const win_stacktrace_entry &) = default;
60
61 constexpr win_stacktrace_entry &
62 operator=(const win_stacktrace_entry &) = default;
63
64 constexpr native_handle_type native_handle() const noexcept { return addr_; }
65
66 constexpr explicit operator bool() const noexcept { return !!addr_; }
67
68 GPCL_DECL std::string description() const;
69
70 GPCL_DECL std::string source_file() const;
71
72 GPCL_DECL std::string binary_file() const;
73
74 GPCL_DECL std::uint_least32_t source_line() const;
75
76 friend inline bool operator<(const win_stacktrace_entry &x,
77 const win_stacktrace_entry &y)
78 {
79 return x.addr_ < y.addr_;
80 }
81
82 friend inline bool operator!=(const win_stacktrace_entry &x,
83 const win_stacktrace_entry &y)
84 {
85 return x.addr_ != y.addr_;
86 }
87
88 friend inline bool operator==(const win_stacktrace_entry &x,
89 const win_stacktrace_entry &y)
90 {
91 return !(x != y);
92 }
93
94 template <typename CharT, typename Traits>
95 friend std::basic_ostream<CharT, Traits> &
96 operator<<(std::basic_ostream<CharT, Traits> &os,
97 const win_stacktrace_entry &f)
98 {
99#if defined(GPCL_BFD) || defined(__CYGWIN__)
100 return os << f.description();
101#else
102 typename std::basic_ostream<CharT, Traits>::sentry sentry(os);
103 if (!sentry)
104 return os;
105
106 if (!f)
107 return os << "[empty]";
108
109 auto desc = f.description();
110 auto file = f.source_file();
111 auto line = f.source_line();
112
113 if (desc.empty())
114 os << f.native_handle();
115 else
116 os << desc;
117
118 if (line)
119 return os << " at " << file << '(' << std::dec << line << ')';
120
121 auto binary = f.binary_file();
122 if (!binary.empty())
123 os << " in " << binary;
124
125 return os;
126#endif
127 }

254template <typename CharT, typename Traits, typename Allocator>
255std::basic_ostream<CharT, Traits> &
256operator<<(std::basic_ostream<CharT, Traits> &os,
257 const basic_win_stacktrace<Allocator> &st)
258{
259 typename std::basic_ostream<CharT, Traits>::sentry sentry(os);
260 if (!sentry)
261 return os;
262
263 os << '\n';
264
265 int i(0);
266 for (auto &e : st)
267 {
268 os << std::setw(4) << std::dec << ++i << "# " << e << '\n';
269 }
270
271 return os;
272}
