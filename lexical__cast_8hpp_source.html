117template <typename CharType, typename Target>
118void lexical_cast_extract_result(std::basic_istream<CharType> &is,
119 Target &result)
120{
121 if (is)
122 {
123 is >> result;
124
125 if (is && is.eof())
126 return;
127 }
128
129 GPCL_THROW(bad_lexical_cast());
130}

132template <typename Target>
133struct lexical_cast_impl
134{
135 static_assert(std::is_default_constructible_v<Target>);
136 static_assert(std::is_copy_constructible_v<Target>);
137
138 template <typename Source, bool False = false>
139 Target operator()(const Source &arg) const
140 {
141 Target result{};
142
143 if constexpr (is_output_streamable<Source, std::ostream>() &&
144 is_input_streamable<Target, std::istream>())
145 {
146 lexical_cast_istream<char, Source> is(arg);
147 lexical_cast_extract_result(is, result);
148 }
149
150 else if constexpr (is_output_streamable<Source, std::wostream>() &&
151 is_input_streamable<Target, std::wistream>())
152 {
153 lexical_cast_istream<wchar_t, Source> is(arg);
154 lexical_cast_extract_result(is, result);
155 }
156
157 else
158 {
159 GPCL_UNREACHABLE("cannot cast such type");
160 }
161
162 return result;
163 }
164
165 template <typename CharType>
166 Target operator()(const CharType *s, std::size_t n) const
167 {
168 Target result{};
169 basic_ispanstream<CharType> is(
170 span<CharType>(const_cast<CharType *>(s), n));
171 lexical_cast_extract_result(is, result);
172 return result;
173 }
174};
