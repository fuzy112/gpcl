22template <typename T>
23struct default_clone
24{
25 T *operator()(const T *value) const
26 {
27 if (value)
28 return ::new T(*value);
29 return nullptr;
30 }
31};

41template <typename T, typename Cloner = default_clone<T>,
42 typename Deleter = default_delete<T>>
43class clone_ptr

87 void swap(clone_ptr &other) noexcept

89 using std::swap;

91 swap(pair_.first(), other.pair_.first());

95 deleter get_deleter() const noexcept { return pair_.first().get_deleter(); }
96
97 cloner get_cloner() const noexcept { return pair_.second(); }
98
99 T *operator->() noexcept { return pair_.first().get(); }
100
101 const T *operator->() const noexcept { return pair_.first().get(); }
102
103 T &operator*() noexcept { return *operator->(); }

105 const T &operator*() const noexcept { return *operator->(); }

107private:
