24template <typename OutIt, typename CharType = char,
25 typename CharTraits = std::char_traits<CharType>>
26class output_iterator_streambuf
27 : public std::basic_streambuf<CharType, CharTraits>
28{
29 using base_type = std::basic_streambuf<CharType, CharTraits>;
30
31public:
32 using char_type = CharType;
33 using int_type = typename CharTraits::int_type;
34
35 explicit output_iterator_streambuf(OutIt out) : out_(out) {}
36
37 OutIt out()
38 {
39 // sync() never fails
40 output_iterator_streambuf::sync();
41 return out_;
42 }
43
44protected:
45 output_iterator_streambuf *setbuf(char_type *s, std::streamsize n) override
46 {
47 base_type::setp(s, s + n);
48 return this;
49 }
50
51 int_type overflow(int_type ch = CharTraits::eof()) override
52 {
53 out_ = std::copy(base_type::pbase(), base_type::pptr(), out_);
54 if (ch != CharTraits::eof())
55 {
56 *out_++ = CharTraits::to_char_type(ch);
57 }
58 base_type::setp(base_type::pbase(), base_type::epptr());
59 return 0;
60 }
61
62 int sync() override
63 {
64 if (output_iterator_streambuf::overflow() == CharTraits::eof())
65 return -1; // should never happens;
66 return 0;
67 }
68
69private:
70 OutIt out_;
71};
