49/* tlsf_t: a TLSF structure. Can contain 1 to N pools. */
50/* pool_t: a block of memory that TLSF can manage. */
51typedef void *tlsf_t;
52typedef void *pool_t;
53
54/* Create/destroy a memory pool. */
55GPCL_DECL tlsf_t tlsf_create(void *mem);
56GPCL_DECL tlsf_t tlsf_create_with_pool(void *mem, size_t bytes);
57GPCL_DECL void tlsf_destroy(tlsf_t tlsf);
58GPCL_DECL pool_t tlsf_get_pool(tlsf_t tlsf);
59
60/* Add/remove memory pools. */
61GPCL_DECL pool_t tlsf_add_pool(tlsf_t tlsf, void *mem, size_t bytes);
62GPCL_DECL void tlsf_remove_pool(tlsf_t tlsf, pool_t pool);
63
64/* malloc/memalign/realloc/free replacements. */
65GPCL_DECL void *tlsf_malloc(tlsf_t tlsf, size_t bytes);
66GPCL_DECL void *tlsf_memalign(tlsf_t tlsf, size_t align, size_t bytes);
67GPCL_DECL void *tlsf_realloc(tlsf_t tlsf, void *ptr, size_t size);
68GPCL_DECL void tlsf_free(tlsf_t tlsf, void *ptr);
69
70/* Returns internal block size, not original request size */
71GPCL_DECL size_t tlsf_block_size(void *ptr);
72
73/* Overheads/limits of internal structures. */
74GPCL_DECL size_t tlsf_size(void);
75GPCL_DECL size_t tlsf_align_size(void);
76GPCL_DECL size_t tlsf_block_size_min(void);
77GPCL_DECL size_t tlsf_block_size_max(void);
78GPCL_DECL size_t tlsf_pool_overhead(void);
79GPCL_DECL size_t tlsf_alloc_overhead(void);
80
81/* Debugging. */
82typedef void (*tlsf_walker)(void *ptr, size_t size, int used, void *user);
83GPCL_DECL void tlsf_walk_pool(pool_t pool, tlsf_walker walker, void *user);
84/* Returns nonzero if any internal consistency check fails. */
85GPCL_DECL int tlsf_check(tlsf_t tlsf);
86GPCL_DECL int tlsf_check_pool(pool_t pool);
