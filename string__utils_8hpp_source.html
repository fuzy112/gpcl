34template <typename BidIt>
35class sub_string : public std::pair<BidIt, BidIt>
36{
37public:
38 using difference_type = typename std::iterator_traits<BidIt>::difference_type;
39 using iterator = BidIt;
40 using value_type = typename std::iterator_traits<BidIt>::value_type;
41
42 difference_type length() const
43 {
44 return std::distance(this->first, this->second);
45 }
46
47 std::basic_string<value_type> str() const
48 {
49 return std::basic_string<value_type>(this->first, this->second);
50 }
51
52 operator std::basic_string<value_type>() const { return str(); }
53
54 bool empty() const { return this->first == this->second; }
55
56 sub_string trimmed() const
57 {
58 auto i = std::find_if(this->first, this->second,
59 [](auto ch) { return !std::isspace(ch); });
60
61 auto j =
62 std::find_if(i, this->second, [](auto ch) { return std::isspace(ch); });
63 return sub_string{{i, j}};
64 }

178template <typename BidIt, typename Alloc, typename Elem>
179void split(BidIt first, BidIt last, split_results<BidIt, Alloc> &result,
180 const Elem *delim, split_flags flag = split_flags())
181{
182 detail::split_results_builder<BidIt, Alloc> rb(result, first);
183 const Elem *delim_end = delim;
184 while (*delim_end != Elem())
185 ++delim_end;
186
187 for (;;)
188 {
189 auto pos = std::search(first, last, delim, delim_end);
190 gpcl::sub_string<BidIt> substr{{first, pos}};
191 if (!substr.empty() || !flag.is_set(split_flag::skip_empty_string))
192 {
193 if (flag & split_flag::trim_results)
194 rb.add(substr.trimmed());
195 else
196 rb.add(substr);
197 }
198 if (pos == last)
199 break;
200
201 first = pos + (delim_end - delim);
202 }
203}

205template <typename BidIt, typename Alloc, typename Elem>
206void split_old(BidIt first, BidIt last, split_results<BidIt, Alloc> &result,
207 const Elem *delim, split_flags flag = split_flags())
208{
209 detail::split_results_builder<BidIt, Alloc> rb(result, first);
210
211 const std::size_t delim_length = std::strlen(delim);
212 auto full_length = std::distance(first, last);
213
214 const Elem *delim_end = delim + delim_length;
215
216 BidIt substr_start = first;
217
218 while (full_length >= delim_length)
219 {
220 if (std::equal(delim, delim_end, first))
221 {
222 const gpcl::sub_string<BidIt> substr{{substr_start, first}};
223 if (!substr.empty() || !flag.is_set(split_flag::skip_empty_string))
224 {
225 rb.add(substr);
226 }
227 first += delim_length;
228 substr_start = first;
229 full_length -= delim_length;
230 continue;
231 }
232 ++first;
233 --full_length;
234 }
235
236 const gpcl::sub_string<BidIt> substr{{substr_start, last}};
237 if (!substr.empty() || !flag.is_set(split_flag::skip_empty_string))
238 {
239 rb.add(substr);
240 }
241}
