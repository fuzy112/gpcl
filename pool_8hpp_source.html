65template <typename UA = default_new_delete_user_allocator,
66 typename MutexType = gpcl::mutex>
67class pool
68{
69 static_assert(is_user_allocator<UA>::value, "user allocator");
70
71public:
72 using user_allocator_type = UA;
73 using mutex_type = MutexType;
74 using size_type = typename user_allocator_type::size_type;
75 using difference_type = typename user_allocator_type::difference_type;
76
78 explicit pool(size_type requested_size)
79 : requested_size_(requested_size),
80 chunk_size_(std::lcm(requested_size_,
81 std::lcm(sizeof(void *), sizeof(size_type)))),
82 block_list_(),
83 next_size_(32)
84 {
85 GPCL_ASSERT(requested_size != 0);
86 }
87
91 ~pool() noexcept
92 {
93 block_info block = block_list_;
94 while (block.ptr)
95 {
96 auto next = *reinterpret_cast<block_info *>(
97 block.ptr + block_info_offset(block.size));
98 user_allocator_type::free(block.ptr);
99 block = next;
100 }
101 }
102
104 size_type requested_size() const noexcept { return requested_size_; }
105
107 [[nodiscard]] void *malloc()
108 {
109 gpcl::unique_lock<mutex_type> lock(mutex_);
110
111 if (storage_.empty())
112 {
113 if (request_new_block(chunk_size_))
114 return storage_.malloc(chunk_size_);
115 else
116 return nullptr;
117 }
118
119 return storage_.malloc(chunk_size_);
120 }
121
123 [[nodiscard]] void *ordered_malloc()
124 {
125 gpcl::unique_lock<mutex_type> lock(mutex_);
126
127 if (storage_.empty())
128 {
129 if (request_new_block(chunk_size_, true))
130 return storage_.malloc(chunk_size_);
131 else
132 return nullptr;
133 }
134
135 return storage_.malloc(chunk_size_);
136 }
137
139 [[nodiscard]] void *ordered_malloc(size_type n)
140 {
141 GPCL_ASSERT(n != 0);
142 auto count = chunk_count(n);
143 gpcl::unique_lock<mutex_type> lock(mutex_);
144 if (auto ret = storage_.malloc_n(count, chunk_size_))
145 return ret;
146 request_new_block(count * chunk_size_, true);
147
148 return storage_.malloc_n(count, chunk_size_);
149 }
150
152 void free(void *ptr)
153 {
154 gpcl::unique_lock<mutex_type> lock(mutex_);
155 storage_.free(ptr);
156 }
157
159 void ordered_free(void *ptr)
160 {
161 gpcl::unique_lock<mutex_type> lock(mutex_);
162 storage_.ordered_free(ptr);
163 }
164
166 void ordered_free(void *ptr, size_type n)
167 {
168 gpcl::unique_lock<mutex_type> lock(mutex_);
169 storage_.ordered_free_n(ptr, chunk_count(n), chunk_size_);
170 }
171
172 // not thread-safe
173 size_type get_next_size() const { return next_size_; }
174
175 // not thread-safe
176 void set_next_size(size_type next_size)
177 {
178 GPCL_ASSERT(next_size != 0);
179 next_size_ = next_size;
180 }
181
182 bool is_from(void *ptr) const noexcept
183 {
184 block_info block = block_list_;
185 while (block.ptr)
186 {
187 if (ptr >= block.ptr && ptr < block.ptr + block_info_offset(block.size))
188 return (reinterpret_cast<char *>(ptr) - block.ptr) % chunk_size_ == 0;
189
190 block = *reinterpret_cast<block_info *>(block.ptr +
191 block_info_offset(block.size));
192 }
193
194 return false;
195 }
196
197protected:
198 struct block_info
199 {
200 // Pointer to the start of the next block.
201 char *ptr;
202
203 // Size of the next block.
204 size_type size;
205 }
206#ifdef __GNUC__
207 __attribute__((may_alias))
208#endif
209 ;
210
211 // Requests a block with at least min_size bytes for chunks.
212 bool request_new_block(size_type min_size, bool ordered = false)
213 {
214 auto sz = next_size_ * chunk_size_;
215 while (sz < min_size + sizeof(block_info) * 2)
216 sz *= 2;
217
218 next_size_ *= 2;
219
220 char *block = user_allocator_type::malloc(sz);
221 if (block == nullptr)
222 return false;
223
224 block_list_ = add_block(block, sz, block_list_, ordered);
225 GPCL_VERIFY_FALSE(storage_.empty());
226 return true;
227 }
228
229 block_info add_block(char *ptr, size_type sz, block_info end, bool ordered)
230 {
231 auto chunks_size = block_info_offset(sz);
232 if (ordered)
233 storage_.add_ordered_block(ptr, chunks_size, chunk_size_);
234 else
235 storage_.add_block(ptr, chunks_size, chunk_size_);
236 *reinterpret_cast<block_info *>(ptr + chunks_size) = end;
237 return {ptr, sz};
238 }
239
240 static difference_type block_info_offset(size_type block_size)
241 {
242 return (block_size - sizeof(block_info)) / alignof(block_info) *
243 alignof(block_info);
244 }
245
246 size_type chunk_count(size_type n)
247 {
248 const size_type array_size = n * requested_size_;
249 const size_type ret = (array_size + chunk_size_ - 1) / chunk_size_;
250 GPCL_VERIFY(ret * chunk_size_ >= n * requested_size_);
251 return ret;
252 }
253
254 mutable mutex_type mutex_;
255 const size_type requested_size_;
256 const size_type chunk_size_;
257 size_type next_size_;
258 block_info block_list_;
259 simple_segregated_storage<size_type> storage_;
260};
