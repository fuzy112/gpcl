30template <typename CharType, typename Traits = std::char_traits<CharType>,
31 typename Allocator = std::allocator<CharType>>
32class basic_syncbuf : public std::basic_streambuf<CharType, Traits>
33{
34public:
35 using allocator_type = Allocator;
36 using streambuf_type = std::basic_streambuf<CharType, Traits>;
37
38private:
39 // state flags.
40
41 enum flags
42 {
43 emit_on_sync,
44 pending_flush,
45
46 n_flags,
47 };
48
49 // the wrapped streambuf.
50 std::basic_streambuf<CharType, Traits> *wrapped_ = nullptr;
51
52 std::bitset<n_flags> flags_;
53
54 // internal buffer for temporary storage.
55 std::basic_string<CharType, Traits, Allocator> buffer_;
56
57 // mutex which protects the wrapped buffer.
58 detail::mutex_for_address_ptr mutex_ =
59 detail::get_mutex_for_address(wrapped_);
60
61public:
63 basic_syncbuf() : basic_syncbuf(nullptr) {}
64
68 explicit basic_syncbuf(streambuf_type *streambuf,
69 const Allocator &alloc = Allocator())
70 : wrapped_(streambuf),
71 buffer_(alloc)
72 {
73 }
74
78 basic_syncbuf(basic_syncbuf &&other) noexcept
79 : streambuf_type(other),
80 wrapped_(other.wrapped_),
81 flags_(other.flags_),
82 buffer_(std::move(other).buffer_),
83 mutex_(std::move(other).mutex_)
84 {
85 other.wrapped_ = false;
86 other.flags_.reset();
87 other.mutex_ = nullptr;
88 }
89
93 basic_syncbuf &operator=(basic_syncbuf &&other) noexcept
94 {
95 emit();
96
97 streambuf_type::operator=(std::move(other));
98 wrapped_ = other.wrapped_;
99 flags_ = other.flags_;
100 buffer_ = std::move(other).buffer_;
101 mutex_ = std::move(other).mutex_;
102
103 other.wrapped_ = nullptr;
104 other.flags_.reset();
105 other.mutex_ = nullptr;
106
107 return *this;
108 }
109
111 void swap(basic_syncbuf &other) noexcept
112 {
113 streambuf_type::swap(other);
114 swap(wrapped_, other.wrapped_);
115 swap(flags_, other.flags_);
116 swap(buffer_, other.buffer_);
117 swap(mutex_, other.mutex_);
118 }
119
123 ~basic_syncbuf()
124 {
125 GPCL_TRY { emit(); }
126 GPCL_CATCH(...) {}
127 GPCL_CATCH_END
128 }
129
132 bool emit()
133 {
134 if (!wrapped_)
135 return false;
136
137 gpcl::scoped_lock<mutex> lock(*mutex_);
138 auto n = wrapped_->sputn(buffer_.data(), buffer_.size());
139 buffer_.erase(buffer_.begin(), buffer_.begin() + n);
140
141 if (flags_.test(pending_flush))
142 {
143 flags_.reset(pending_flush);
144 if (wrapped_->pubsync() != 0)
145 return false;
146 }
147
148 return buffer_.empty();
149 }
150
152 streambuf_type *get_wrapped() const noexcept { return wrapped_; }
153
155 allocator_type get_allocator() const noexcept
156 {
157 return buffer_.get_allocator();
158 }
159
161 void set_emit_on_sync(bool b) noexcept
162 {
163 flags_.set(emit_on_sync, b);
164 }
165
166protected:
169 int sync() override
170 {
171 flags_.set(pending_flush);
172
173 GPCL_TRY
174 {
175 if (flags_.test(emit_on_sync))
176 return emit() ? 0 : -1;
177 }
178 GPCL_CATCH(...) { return -1; }
179 GPCL_CATCH_END
180 return 0;
181 }
182
183 using int_type = typename Traits::int_type;
184
185 GPCL_CLANG_SUPPRESS_WARNING_WITH_PUSH("-Wconstant-conversion")
186
187
188 int_type overflow(int_type ch = Traits::eof()) override
189 {
190 GPCL_TRY
191 {
192 if (Traits::eq_int_type(ch, Traits::eof()))
193 return ~Traits::eof();
194
195 buffer_.push_back(Traits::to_char_type(ch));
196 return ~Traits::eof();
197 }
198 GPCL_CATCH(...) { return Traits::eof(); }
199 GPCL_CATCH_END
200 return 0;
201 }
202
203 GPCL_CLANG_SUPPRESS_WARNING_POP
204
205 std::streamsize xsputn(const CharType *s,
206 std::streamsize count) noexcept override
207 {
208 GPCL_TRY
209 {
210 buffer_.append(s, count);
211 return count;
212 }
213 GPCL_CATCH(...) { return 0; }
214 GPCL_CATCH_END
215 }
216};
