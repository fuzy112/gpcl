34template <typename To, typename From>
35constexpr To narrow_cast(From F) noexcept
36{
37 static_assert(std::is_arithmetic<From>() && std::is_arithmetic<To>(),
38 "Both From and To must be arithmetic type");

40 To T = static_cast<To>(F);
