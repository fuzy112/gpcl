28offset_ptr<T> &offset_ptr<T>::operator=(const offset_ptr &other)

51template <typename T>
52
53typename offset_ptr<T>::pointer offset_ptr<T>::get() const
54{
55 GPCL_CXX17_IF_CONSTEXPR (!std::is_void<value_type>::value &&
56 !std::is_same<value_type, char>::value &&
57 !std::is_same<value_type, signed char>::value &&
58 !std::is_same<value_type, unsigned char>::value
59#if __cplusplus > 201703L
60 && !std::is_same<value_type, std::byte>::value
61#endif
62 )
63 {
64 GPCL_ASSERT(offset_ != 0); // maybe using uninitialized value
65 }
66 return offset_ != invalid_offset
67 ? reinterpret_cast<pointer>(detail::as_integer(this) + offset_)
68 : nullptr;
69}
