63 synchronised_pool_resource(pool_options const &opts,

75 ~synchronised_pool_resource()

79 upstream_->deallocate(std::get<0>(block), std::get<1>(block),

81 }

84 void release() noexcept
85 {
86 scoped_lock lock(mtx_);
87 pools_.clear();
88
89 for (auto block : blocks_from_upstream_)
90 {
91 upstream_->deallocate(std::get<0>(block), std::get<1>(block),
92 std::get<2>(block));
93 }
94 blocks_from_upstream_.clear();

96 create_pools();
97 }

103protected:
104 void *do_allocate(std::size_t bytes, std::size_t alignment) override
105 {
106 if (bytes >= options_.largest_required_pool_block ||
107 alignment > alignof(std::max_align_t))
108 {
109 return upstream_->allocate(bytes, alignment);
110 }
111
112 scoped_lock lock(mtx_);
113 return allocate_from_pool(bytes, alignment);
114 }
115
116 void do_deallocate(void *ptr, std::size_t bytes,
117 std::size_t alignment) noexcept override
118 {
119 if (bytes >= options_.largest_required_pool_block ||
120 alignment > alignof(std::max_align_t))
121 {
122 return upstream_->deallocate(ptr, bytes, alignment);
123 }
124
125 scoped_lock lock(mtx_);
126 deallocate_from_pool(ptr, bytes, alignment);
127 }
128
129 bool do_is_equal(const memory_resource &other) const noexcept override
130 {
131 return this == &other;
132 }
133
134 void *allocate_from_pool(std::size_t bytes, std::size_t alignment)
135 {
136 pool_type &pool = get_pool(bytes, alignment);
137 GPCL_ASSERT(bytes % pool.requested_size() == 0);
138 return pool.ordered_malloc(bytes / pool.requested_size());
139 }
140
141 void deallocate_from_pool(void *ptr, std::size_t bytes,
142 std::size_t alignment) noexcept
143 {
144 pool_type &pool = get_pool(bytes, alignment);
145 GPCL_ASSERT(bytes % pool.requested_size() == 0);
146 pool.ordered_free(ptr, bytes / pool.requested_size());
147 }
148
149 pool_type &get_pool(std::size_t bytes, std::size_t alignment) noexcept
150 {
151 (void)alignment;
152 for (auto iter = pools_.rbegin(), end = pools_.rend(); iter != end; ++iter)
153 {
154 if (bytes % iter->first == 0)
155 {
156 return iter->second;
157 }
158 }
159 GPCL_UNREACHABLE("impossible");
160 }
161
162 void create_pools()
163 {
164 std::size_t pool_sizes[] = {
165 1, 2, 4, 8, 16, 32, 64, 96, 128, 256,
166 };
167
168 for (auto const size : pool_sizes)
169 {
170 if (size >= options_.largest_required_pool_block)
171 break;
172 pools_.try_emplace(size, size, upstream_);
173 }
174 }
175};
