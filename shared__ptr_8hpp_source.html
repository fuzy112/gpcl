263 template <typename Allocator, typename... Args>
264 shared_ptr(in_place_t, const Allocator &alloc, Args &&...args)
265 {
266 auto *rc_obj = detail::ref_count_obj<T, Allocator>::create(
267 alloc, std::forward<Args>(args)...);
268 p_ = rc_obj->get_address();
269 s_ = rc_obj;
270
271 this->enables_shared_from_this(p_);
272 }
273
275 ~shared_ptr()
276 {
277 if (s_)
278 s_->put();
279 s_ = nullptr;
280 p_ = nullptr;
281 }
