84 void span(span<CharType> buf) noexcept
85 {
86 span_ = buf;
87
88 CharType *begin = span_.data();
89 CharType *end = begin + span_.size();
90
91 if (openmode_ & std::ios_base::in)
92 {
93 this->setg(begin, begin, end);
94 }
95
96 if (openmode_ & std::ios_base::out)
97 {
98 if (openmode_ & std::ios_base::ate)
99 {
100 this->setp(end, end);
101 }
102 else
103 {
104 this->setp(begin, end);
105 }
106 }
107 }

127 pos_type seekoff(off_type off, std::ios_base::seekdir dir,
128 std::ios_base::openmode which = std::ios_base::in |
129 std::ios_base::out) override
130 {
131 if ((openmode_ & which) != which)
132 return off_type(-1);
133
134 off_type newoff;
135 switch (dir)
136 {
137 case std::ios_base::beg:
138 newoff = off;
139 break;
140
141 case std::ios_base::cur:
142 if ((which & std::ios_base::in) && (which & std::ios_base::out))
143 return off_type(-1);
144
145 if (which & std::ios_base::out)
146 newoff = this->pptr() - this->pbase() + off;
147 else if (which & std::ios_base::in)
148 newoff = this->gptr() - this->eback() + off;
149 else
150 return off_type(-1);
151 break;
152
153 case std::ios_base::end:
154 if ((openmode_ & std::ios_base::out) && !(openmode_ & std::ios_base::in))
155 newoff = this->pptr() - this->pbase() + off;
156 else
157 newoff = off + span_.size();
158 break;
159
160 default:
161 GPCL_UNREACHABLE("invalid seekdir");
162 }
163
164 if (this->pptr() == nullptr && this->gptr() == nullptr && newoff != 0)
165 return off_type(-1);
166
167 if (newoff < 0 || std::size_t(newoff) > span_.size())
168 return off_type(-1);
169
170 CharType *pbuf = span_.data();
171 if (which & std::ios_base::in)
172 this->setg(pbuf, pbuf + newoff, pbuf + span_.size());
173
174 if (which & std::ios_base::out)
175 this->setp(pbuf + newoff, pbuf + span_.size());
176
177 return newoff;
178 }
179
180 pos_type

182 std::ios_base::openmode which = std::ios_base::in |
183 std::ios_base::out) override
184 {
185 return seekoff(off_type(pos), std::ios_base::beg, which);
186 }
